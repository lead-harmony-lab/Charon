14class QuartermasterPayload(BaseAgentPayload): 

15 """Payload for logistics and PartVault database operations."""

42 @model_validator(mode="before") 

43 @classmethod 

44 def sanitize_quartermaster_payload(cls, data: Any) -> Any: 

45 if not isinstance(data, dict): 

46 return data 

47 

48 mpn_candidate = data.get("mpn") or data.get("part_number") or data.get("query") 

49 if mpn_candidate: 

50 if not data.get("part_number"): 

51 data["part_number"] = mpn_candidate 

52 if not data.get("mpn"): 

53 data["mpn"] = mpn_candidate 

54 return data
