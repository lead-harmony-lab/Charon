16 

17class CapabilityContract(StrictBaseModel): 

18 """Declarative contract defining what an agent capability requires and produces.""" 

19 

20 capability_name: str = Field( 

21 description="Unique identifier for the capability (e.g., 'fetch_datasheet', 'launch_gui_viewer')." 

22 ) 

23 agent: AgentEnum = Field( 

24 description="The specialist agent providing this capability." 

25 ) 

26 description: str = Field( 

27 description="Clear operational summary of what this capability accomplishes." 

28 ) 

29 escalation_level: EscalationLevel = Field( 

30 default=EscalationLevel.L1_SPECIALIST, 

31 description="The escalation tier at which this capability operates.", 

32 ) 

33 consumed_artifacts: List[str] = Field( 

34 default_factory=list, 

35 description="Blackboard artifact keys required as input (e.g., ['target_part'], ['resolved_file_path']).", 

36 ) 

37 produced_artifacts: List[str] = Field( 

38 default_factory=list, 

39 description="Blackboard artifact keys guaranteed/expected to be generated (e.g., ['resolved_file_path']).", 

40 ) 

41 required_binaries: List[str] = Field( 

42 default_factory=list, 

43 description="Optional OS binaries required to execute this capability (e.g., ['evince', 'xdg-open']).", 

44 )

150# --- Capability Helper Utility Functions --- 

151 

152def get_capability(capability_name: str) -> Optional[CapabilityContract]: 

153 """Retrieves a capability contract by name.""" 

154 return CAPABILITY_REGISTRY.get(capability_name) 

155 

156 

157def find_capabilities_for_artifacts(available_artifacts: Set[str]) -> List[CapabilityContract]: 

158 """Finds all capabilities whose consumed artifact prerequisites are fully met on the blackboard.""" 

159 matching = [] 

160 for cap in CAPABILITY_REGISTRY.values(): 

161 if set(cap.consumed_artifacts).issubset(available_artifacts): 

162 matching.append(cap) 

163 return matching 

164 

165 

166def find_capabilities_by_agent(agent: AgentEnum) -> List[CapabilityContract]: 

167 """Lists all capabilities registered under a specific agent.""" 

168 return [cap for cap in CAPABILITY_REGISTRY.values() if cap.agent == agent] 

169 

170 

171def find_capabilities_by_escalation(level: EscalationLevel) -> List[CapabilityContract]: 

172 """Lists all capabilities operating at a given escalation level.""" 

173 return [cap for cap in CAPABILITY_REGISTRY.values() if cap.escalation_level == level]
