98def audit_report(skills: List[Dict]): 

99 """Displays a detailed diagnostic health matrix across all skills.""" 

100 console.clear() 

101 table = Table(title="Diagnostic System Audit", show_header=True, header_style="bold cyan") 

102 table.add_column("Skill ID", style="bold white") 

103 table.add_column("Stage", style="blue") 

104 table.add_column("Status", style="bold") 

105 table.add_column("Missing Binaries", style="yellow") 

106 table.add_column("APT Package Mapping", style="magenta") 

107 

108 for s in skills: 

109 missing = s.get("missing_requirements", []) 

110 if missing: 

111 status = "[bold red]CRITICAL[/bold red]" 

112 packages = [PACKAGE_MAP.get(m, m) for m in missing] 

113 table.add_row( 

114 s["skill_id"], 

115 s["stage"], 

116 status, 

117 ", ".join(missing), 

118 ", ".join(packages), 

119 ) 

120 else: 

121 table.add_row( 

122 s["skill_id"], 

123 s["stage"], 

124 "[bold green]HEALTHY[/bold green]", 

125 "[dim]None[/dim]", 

126 "[dim]N/A[/dim]", 

127 ) 

128 

129 console.print(table) 

130 Prompt.ask("\nPress Enter to return to Diagnostics Menu")

133def resolve_all_dependencies(broken_skills: List[Dict]): 

134 """Collects missing requirements, applies package mapping, and triggers apt-get.""" 

135 console.clear() 

136 

137 # Collect all missing binaries across all broken skills 

138 missing_binaries = set() 

139 for s in broken_skills: 

140 missing_binaries.update(s.get("missing_requirements", [])) 

141 

142 # Translate binary names to APT packages 

143 apt_packages = [PACKAGE_MAP.get(b, b) for b in missing_binaries] 

144 pkg_str = " ".join(apt_packages) 

145 

146 cmd = f"sudo apt-get update && sudo apt-get install -y {pkg_str}" 

147 

148 console.print("[bold red]⚠️ DEPENDENCY RESOLUTION TARGETS DETECTED[/bold red]\n") 

149 console.print(f" [bold]Missing Binaries ($PATH):[/bold] {', '.join(missing_binaries)}") 

150 console.print(f" [bold]Target APT Packages:[/bold] [cyan]{pkg_str}[/cyan]") 

151 console.print(f" [bold]Execution Command:[/bold] [dim]{cmd}[/dim]\n") 

152 

153 confirm = Prompt.ask("Execute package installation with elevated privileges?", choices=["y", "n"], default="y") 

154 

155 if confirm.lower() == "y": 

156 subprocess.run(cmd, shell=True) 

157 Prompt.ask("\nPress Enter to return and refresh diagnostic health state")
